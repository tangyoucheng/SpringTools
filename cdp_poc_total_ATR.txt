# =========================================================================
# 1. 建立 A股 专属的单日分时自适应 ATR 矩阵
# =========================================================================
intraday_ratios = {
    "p_05": 0.05,   # 细网格：捕捉高频瞬时多空换手
    "p_12": 0.125,  # 标准网格：筹码骨架
    "p_20": 0.20,   # 防噪粗网格：1/5 日内波幅过滤毛刺（作为审计锚定基准）
    "p_40": 0.40    # 宏观大本营
}

def calculate_advanced_poc(df_today, yesterday_atr):
    """
    【终极防弹 3.0 完美形态】当日分时自适应 POC 计算与智能审计系统
    """
    if df_today.empty or len(df_today) < 1:
        return (df_today["Close"].iloc[-1] if not df_today.empty else None), "⚠️ 数据不足"

    # 清洗数据
    df_clean = df_today.dropna(subset=["Open", "High", "Low", "Close", "Volume"]).copy()
    if df_clean.empty:
        return df_today["Close"].iloc[-1], "⚠️ 清洗后无有效数据"

    # 计算 4 大矩阵结果及对应的 bins 结构
    poc_matrix = {}
    bins_matrix = {}
    counts_matrix = {}
    
    for key, atr_multiplier in intraday_ratios.items():
        poc_val, bins_arr, counts_arr = run_intraday_perfect_grid(df_clean, atr_multiplier, yesterday_atr)
        poc_matrix[key] = poc_val
        bins_matrix[key] = bins_arr
        counts_matrix[key] = counts_arr

    p_05 = poc_matrix["p_05"]
    p_20 = poc_matrix["p_20"]

    # 兜底：如果核心网格计算失败（包含返回 None 的场景），返回收盘价
    if p_05 is None or p_20 is None:
        return round(float(df_clean["Close"].iloc[-1]), 3), "⚠️ 网格初始化失败，采信尾盘收盘价"

    # =========================================================================
    # 4. 🤖 智能裁决系统（3.0 修复版：NumPy 属性拦截与优雅 zip 解包）
    # =========================================================================
    audit_threshold = 0.20 * yesterday_atr  # 严防死守，锁死 20% ATR 的最大容忍偏离度

    if abs(p_05 - p_20) > audit_threshold:
        bins_20 = bins_matrix["p_20"]
        counts_20 = counts_matrix["p_20"]
        
        # ⚡【修补 1：NumPy 规范检查】：采用 Python 布尔判定结合 .size 属性，兼顾可读性与无越界安全
        if counts_20 is None or bins_20 is None or counts_20.size == 0:
            return p_20, f"🚨 拦截异常脉冲！粗网格数据异常或触发安全气囊，强制回滚至粗网格中轴({p_20:.2f}元)"
            
        max_idx_20 = np.argmax(counts_20)
        
        # 确定粗网格锁定下的主战场价格边界（绝对安全区域）
        low_bound = bins_20[max_idx_20]
        high_bound = bins_20[max_idx_20 + 1]
        
        # 调取细网格数据
        bins_05 = bins_matrix["p_05"]
        counts_05 = counts_matrix["p_05"]
        
        # 安全前置判定
        if counts_05 is None or bins_05 is None or counts_05.size == 0:
            return p_20, f"🚨 拦截异常脉冲！细网格阵列为空，强制回滚至粗网格中轴({p_20:.2f}元)"
        
        # ⚡【修补 2：zip 切片优雅解包】：彻底解决 [i+1] 隐性越界炸弹，代码逻辑直接拉满
        valid_indices = []
        # 将 counts 和前后错开 1 格的 bins 打包成三元组迭代器
        for idx, (g_low, g_high) in enumerate(zip(bins_05[:-1], bins_05[1:])):
            # 只要细网格区间与粗网格主战场区间存在交集或切点重合，即视为有效备选细网格索引
            if max(low_bound, g_low) <= min(high_bound, g_high):
                valid_indices.append(idx)
        
        if valid_indices:
            valid_indices = np.array(valid_indices)
            # 在安全的局部重叠范围内，找出筹码最密集的细网格索引
            best_sub_idx = valid_indices[np.argmax(counts_05[valid_indices])]
            final_poc = round(float((bins_05[best_sub_idx] + bins_05[best_sub_idx + 1]) / 2), 3)
            audit_status = f"🚨 拦截脉冲骗线！已在主战场范围[{low_bound:.2f}-{high_bound:.2f}]内完成细网格寻优 -> ({final_poc:.2f}元)"
        else:
            final_poc = p_20  
            audit_status = f"🚨 拦截异常脉冲！未找到几何交集，强制回滚至粗网格中轴({p_20:.2f}元)"
    else:
        final_poc = p_05
        audit_status = f"🟢 多空真实换手共振，未见脉冲骗线，采信高精度极细网格({p_05:.2f}元)"

    return final_poc, audit_status


# =========================================================================
# 【无漏洞核心组件】高频分时日内成交量直方图
# =========================================================================
def run_intraday_perfect_grid(df_clean, atr_multiplier, base_atr):
    try:
        bin_size = base_atr * atr_multiplier
        if bin_size < 0.01:
            bin_size = 0.01

        intraday_high = float(df_clean["High"].max())
        intraday_low = float(df_clean["Low"].min())
        price_range = intraday_high - intraday_low

        # 双道防零锁死。严防一字板或 price_range=0 导致重算步长归零，绝育死循环
        if (price_range / bin_size) > 1000:
            bin_size = price_range / 800.0  
        
        if bin_size < 0.01:
            bin_size = 0.01

        # 注入 1e-5 扰动项，消除 np.arange 浮点数边界截断硬伤
        bins = np.arange(
            intraday_low - bin_size * 5,
            intraday_high + bin_size * 5 + bin_size + 1e-5,
            bin_size,
        )

        # 安全气囊返回值对齐：遇到不可算极端状态，counts 返回标准的空阵列
        if len(bins) < 2:
            return round(float((intraday_high + intraday_low) / 2), 3), np.array([intraday_low, intraday_high]), np.array([])

        # 采用高频量化“典型价”单点投射法，封杀上下影线虚空吸量
        typical_prices = ((df_clean["High"] + df_clean["Low"] + df_clean["Close"]) / 3.0).to_numpy().flatten()
        volumes = df_clean["Volume"].to_numpy().flatten()

        # A股集合竞价与极端缩量一字板特殊过滤
        is_auction = (df_clean["Open"] == df_clean["Close"]) & (df_clean["High"] == df_clean["Low"])
        volumes_adjusted = np.where(is_auction, volumes * 0.3, volumes) 

        # 运行直方图频数统计
        counts, _ = np.histogram(typical_prices, bins=bins, weights=volumes_adjusted)
        
        if counts.sum() == 0:
            center = round(float((intraday_high + intraday_low) / 2), 3)
            return center, bins, counts
            
        max_idx = np.argmax(counts)
        
        if max_idx >= len(counts) or (max_idx + 1) >= len(bins):
            center = round(float((intraday_high + intraday_low) / 2), 3)
            return center, bins, counts

        poc_value = round(float((bins[max_idx] + bins[max_idx + 1]) / 2), 3)
        return poc_value, bins, counts

    except Exception as e:
        print(f" 🚨 run_intraday_perfect_grid 运行异常: {e}")
        return None, None, None

if __name__ == "__main__":
    #target_stock = "688037.SS"  # 以芯源微为例
    for stock in stocks_list:
        stock_code = str(stock['stockCode'])
        
        # 判断是否已经带了后缀，如果没有，则根据规则拼接
        if not (stock_code.endswith('.SS') or stock_code.endswith('.SZ')):
            # 科创板(688..)和沪市主板(60..)拼接 .SS，其余（如创业板300..）拼接 .SZ
            if stock_code.startswith(('688', '60')):
                target_stock = f"{stock_code}.SS"
            else:
                target_stock = f"{stock_code}.SZ"
        else:
            target_stock = stock_code

        stock_name = stock['stockName']        
            
        print(f"🏁 激活盘中自适应量化扫描，当前监控标的: {target_stock} {stock_name}")
        
        # 1. ⚙️ 计算昨收闭合的 14日历史波动率作为今日网格硬标尺
        #yesterday_atr = fetch_yesterday_stable_atr(target_stock, period_days=14)
        yesterday_atr = stock['atr14D']
        if yesterday_atr is None or np.isnan(yesterday_atr):
            yesterday_atr = 340.0 * 0.04  # 假设的虚拟兜底
            print(f" ⚠️ 盘前未抓到稳定ATR，触发安全熔断：自动采用虚拟 ATR ({yesterday_atr:.2f}元) 锁定今日标尺")
        else:
            print(f" 🟢 历史波动率标尺锁定成功！今日自适应网格基准 14日ATR 为: {yesterday_atr}元")

        # 2. 📡 实时抓取【今天开盘以来截止到此时此刻】最新的 1分钟级 实时分时图数据
        print(f"📡 正在下载今日开盘至今的最新分时流数据...")
        df_today = yf.download(target_stock, period="1d", interval="1m", progress=False, session=session)
        
        if isinstance(df_today.columns, pd.MultiIndex):
            df_today.columns = df_today.columns.get_level_values(0)
            
        if df_today.empty:
            print(f"🚨 错误：未能在盘中抓取到有效的实时分时Tick数据，请检查网络或确认当前是否处于A股交易时段！")
            sys.exit(0)

        # 🚀 调度运行你的终极 POC 算法
        final_poc, audit_status = calculate_advanced_poc(df_today, yesterday_atr)

        # 5. 运行宏观 2倍POC加权 CDP 五关价公式体系 (纯净日内分时原生交互版)
        # 🚨 【绝杀降维改写】利用 to_numpy().flatten() 无条件将其还原为干净的一维数字链
        today_high_arr = df_today["High"].to_numpy().flatten()
        today_low_arr = df_today["Low"].to_numpy().flatten()
        today_close_arr = df_today["Close"].to_numpy().flatten()

        macro_high = today_high_arr.max()
        macro_low = today_low_arr.min() 
        
        # 提取盘中最靠近现在的一分钟收盘价，作为盘中实时变动的“最新现价”
        intraday_latest_price = float(today_close_arr[-1])

        true_cdp = (macro_high + macro_low + 2 * final_poc) / 4
        ah_poc = true_cdp + (macro_high - macro_low)
        nh_poc = 2 * true_cdp - macro_low
        nl_poc = max(0.01, 2 * true_cdp - macro_high)
        al_poc = max(0.01, true_cdp - (macro_high - macro_low))

        # 6. 优雅打印定格监控看板
        print(f"\n" + "⚡"*8 + f" 盘中即时定格看板 | 刷新时钟: {datetime.datetime.now().strftime('%H:%M:%S')} " + "⚡"*8)
        print(f" 监控标的: {target_stock}  {stock_name}| 此时盘中最新现价: {intraday_latest_price:.2f}元")
        print(f" 审计状态: {audit_status}")
        print(f" 🎯 智能审定的当前真实分时 POC (资金重心): {final_poc:.2f}元")
        print(f"---------------------------------------------------------------------------")
        print(f" [追多突破线 AH]: {ah_poc:.2f}元 | [日内高抛压力 NH]: {nh_poc:.2f}元")
        print(f" [分时中轴线 CDP]: {true_cdp:.2f}元")
        print(f" [日内低吸支撑 NL]: {nl_poc:.2f}元 | [做空破位止损 AL]: {al_poc:.2f}元")
        print(f"===========================================================================\n")
        
        # 🤖 信号动态快显
        if intraday_latest_price <= nl_poc:
            print(f"    💡 【实战备忘】: 现价已跌破 NL 线，正处于黄金低吸区，可执行左侧建仓。")
        elif intraday_latest_price >= nh_poc:
            print(f"    💡 【实战备忘】: 现价已突破 NH 线，多头动能逼近上限，可执行高抛锁定利润。")
        else:
            print(f"    💡 【实战备忘】: 当前处于多空拉锯腹地，策略建议：持股观望，静待回踩或突破。")
        print(f"\n\n")
