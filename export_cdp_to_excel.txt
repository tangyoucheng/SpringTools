from openpyxl.styles import Font, PatternFill

# 使用免安装版本时，为了读取CDP_config.py，添加的设定
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from common_number import RoundingStyle
from common_Volume_Profile import generate_ultimate_matrix_data_cn


        

# 1. 事先定义好和模板一模一样的字体（比如：微软雅黑，11号，不加粗）
template_font = Font(name="微软雅黑", size=11, bold=False)

#stock_info = {
#    "股票代码": target_stock,
#    "股票名称": stock_name,
#    "收盘价": RoundingStyle.ROUND.apply(stock_metrics_close_p),
#    "最高价": RoundingStyle.ROUND.apply(stock_metrics_high_p),
#    "最低价": RoundingStyle.ROUND.apply(stock_metrics_low_p),
#    "5日ATR":  ,
#    "日内恐慌激活线":  ,
#    "反弹触发阈值": ,
#}
def export_cdp_to_excel(
    periods_pool,
    results_board,
    stock_info,
    #target_stock,  # 股票代码
    #stock_name,  # 股票名称
    #stock_metrics_close_p,  # 当前最新价/收盘价
    #stock_metrics_high_p,  # 最高价
    #stock_metrics_low_p,  # 最低价
    end_date,
    template_path="export_cdp_poc_stock_temple.xlsm",
):
    """将股票基本数据与CDP/POC数据无缝写入Excel模板，并在A1输出end_date。
    
    【已修改】所有表格数据统一从第二列（B列）开始对齐写入。
    """

    # 1. 定义统一的 CDP 数据表头
    headers = [
        "周期类型",
        "智能审定真POC",
        "中轴(CDP)",
        "止损边界(AL)",
        "低吸支撑(NL)",
        "高抛阻力(NH)",
        "追多边界(AH)",
    ]

    # 2. 循环遍历数据池，将每一行存入二维列表中
    table_rows = []
    for days in periods_pool:
        res = results_board[days]
        if days <= 5:
            name_str = f"{days}天短线历史"
        else:
            days_map = {
                20: "20日线(月度)",
                60: "60日线(季度)",
                120: "120日线(半年)"
            }
            # 匹配不上字典里的 20、60、120 时，默认返回 250 日线
            name_str = days_map.get(days, "250日线(年度)")

        table_rows.append(
            [
                name_str,
                RoundingStyle.ROUND.apply(res["final_poc"]),  # 四舍五入
                RoundingStyle.ROUND.apply(res["true_cdp"]),   # 四舍五入
                RoundingStyle.FLOOR.apply(res["al_poc"]),     # 直接舍去
                RoundingStyle.CEIL.apply(res["nl_poc"]),      # 直接进位
                RoundingStyle.CEIL.apply(res["nh_poc"]),      # 直接进位
                RoundingStyle.FLOOR.apply(res["ah_poc"]),     # 直接舍去
            ]
        )

    # 将二维列表转为 DataFrame 备用
    excel_data = pd.DataFrame(table_rows, columns=headers)
    # 筹码墙
    chips_wall_df = generate_ultimate_matrix_data_cn(stock_info, excel_data)
    
    # ======================
    # 写入Excel模板
    # ======================

    # 加载模板，保留 VBA 宏
    wb = load_workbook(template_path, keep_vba=True)
    ws = wb["Sheet1"]

    # 清空旧数据
    #ws.delete_rows(22, ws.max_row)

    # 💡 核心控制：定义全局起始列（2 代表 B列）
    START_COL = 2
    START_ROW = 31

    # 【第一行第一列】：输出结束日期 end_date（第一行的其余列保持空白）
    # 依然保持在 A1，不受 START_COL 影响
    ws.cell(row=START_ROW, column=START_COL, value=end_date).font = template_font
    START_ROW = START_ROW + 1;

    # 【第二行】：输出追加的股票属性表头（从第二列开始）
    # stock_info.keys() 会自动提取出 ["股票代码", "股票名称", "收盘价", "最高价", "最低价"]
    for col_idx, header_val in enumerate(stock_info.keys(), start=START_COL):
        ws.cell(row=START_ROW, column=col_idx, value=header_val).font = template_font
    START_ROW = START_ROW + 1;

    # 【第三行】：输出对应的实际股票基本数据值（从第二列开始）
    # stock_info.values() 会自动提取出对应的实际数值（如 target_stock, stock_name 等）
    for col_idx, value_val in enumerate(stock_info.values(), start=START_COL):
        ws.cell(row=START_ROW, column=col_idx, value=value_val).font = template_font
    START_ROW = START_ROW + 1;

    # 【第四行】：输出之前的 CDP/POC 数据的字段表头（从第二列开始）
    for col_idx, header in enumerate(excel_data.columns, start=START_COL):
        ws.cell(row=START_ROW, column=col_idx, value=header).font = template_font
    START_ROW = START_ROW + 1;

    # 【第五行及以后】：依次无缝输出具体的 POC 数据行（从第二列开始）
    for row_offset, row in enumerate(
        excel_data.itertuples(index=False), start=START_ROW
    ):        
        START_ROW = row_offset;
        for col_idx, value in enumerate(row, start=START_COL):
            ws.cell(row=row_offset, column=col_idx, value=value).font = template_font

    # 下拉平滑顺延，隔离视觉数据区
    START_ROW += 2  
    
    # 写入筹码墙中文主大报头
    ws.cell(row=START_ROW, column=START_COL, value=str(f"@@@@@@@@@@@@@@@@@@ 📊 {stock_info['股票名称']} 动态自演化立体筹码分布墙 @@@@@@@@@@@@@@@@@@")).font = template_font
    ws.cell(row=START_ROW, column=START_COL).font = Font(name="微软雅黑",bold=True, size=11, color="104E8B")
    START_ROW += 1
    
    #ws.cell(row=START_ROW, column=START_COL, value=str(f"当前收盘价: {stock_info['收盘价']} | 5日ATR: {round(stock_info['5日ATR'], 4)} | 零人工阀门硬核干预")).font = template_font
    #ws.cell(row=START_ROW, column=START_COL).font = Font(name="微软雅黑",italic=True, size=10, color="555555")
    #START_ROW += 2  
    
    # 写入筹码墙三列核心字段表头
    for col_idx, wall_header in enumerate(chips_wall_df.columns, start=START_COL):
        cell = ws.cell(row=START_ROW, column=col_idx, value=str(wall_header)).font = template_font
        cell.font = Font(name="微软雅黑",bold=True)
    START_ROW += 1
    
    # 灌入自演化网格数据
    for row in chips_wall_df.itertuples(index=False):
        for col_idx, value in enumerate(row, start=START_COL):
            # 判断当前单元格数据，处理空白占位或强类型转换
            if value == "":
                ws.cell(row=START_ROW, column=col_idx, value="").font = template_font
            elif isinstance(value, (int, float)):
                ws.cell(row=START_ROW, column=col_idx, value=float(value)).font = template_font
            else:
                ws.cell(row=START_ROW, column=col_idx, value=str(value)).font = template_font
                
            cell = ws.cell(row=START_ROW, column=col_idx)
            
            # Excel专属条件颜色样式渲染
            row_str = str(row)
            if "★" in row_str:
                cell.fill = PatternFill(start_color="FFE4E1", end_color="FFE4E1", fill_type="solid")
                cell.font = Font(name="微软雅黑", size=11,color="8B0000", bold=True)
            elif "🚨" in row_str:
                cell.font = Font(name="微软雅黑", size=11,color="FF6600", italic=True)
                    
        START_ROW += 1
        
    # 生成动态文件名
    # 1. 获取当前日期字符串 (格式如: 2026-07-27)
    #current_date_str = datetime.now().strftime("%Y-%m-%d")

    # 2. 定义目标文件夹路径
    output_dir = Path(end_date)

    # 3. 自动创建文件夹（若文件夹已存在则忽略）
    output_dir.mkdir(parents=True, exist_ok=True)

    # 4. 拼接完整的文件路径
    excel_path = output_dir / f"{stock_info['股票代码']}_{stock_info['股票名称']}_{end_date}.xlsm"
    #excel_path = f"{stock_info["股票代码"]}_{stock_info["股票名称"]}_{end_date}.xlsm"

    # 保存文件
    wb.save(excel_path)
    print(f"{excel_path} 保存完成")
