def fetch_dynamic_atr(df_try, period_days=14):
    """【资产管理级自适应核心组件 - V16 傲视穹穹终定体】

    1. 双向夹逼锁：引入种子期天花板常识锁，彻底封杀“种子期乌龙巨震 + 后续无限一字跌停”引发的 ATR 虚高不退与风控僵死盲区。
    2. 零分发规范：本函数已做极致轻量化标量重整。外层工程部署时，强烈建议在上游一次性将大表重整为物理 C 连续矩阵，
       传入本函数的 df_try["High"].to_numpy() 等应当是原生连续视图，以消灭频繁 malloc 导致的渐进式回测卡顿。
    """
    try:
        # 🚨【V16 规范提示：若上游已固化 C 连续，此处 np.array(..., order='C') 将自动退化为极速 O(1) 视图引用】
        high = np.array(df_try["High"], dtype=float, order="C")
        low = np.array(df_try["Low"], dtype=float, order="C")
        close = np.array(df_try["Close"], dtype=float, order="C")

        total_len = len(high)
        tr_len = total_len - 1

        # 10N 场景标准拦截线
        if tr_len < period_days:
            return None

        # 🚨【SIMD 级全毒素原子化洗净防线】
        if (
            np.isnan(close).any() or np.isnan(high).any() or np.isnan(low).any()
            or np.isinf(close).any() or np.isinf(high).any() or np.isinf(low).any()
        ):
            return None

        # 性能压榨：由于 NaN 已被全量封杀，后续直接呼叫纯正的原生 C 级 np.min 矢量算子
        if np.min(close) <= 0.0 or np.min(high) <= 0.0 or np.min(low) <= 0.0:
            return None

        # 1. 完美的 TR 矩阵化计算
        high_low = high[1:] - low[1:]
        high_close_prev = np.abs(high[1:] - close[:-1])
        low_close_prev = np.abs(low[1:] - close[:-1])
        tr_np = np.maximum(
            high_low, np.maximum(high_close_prev, low_close_prev)
        )

        # 2. 完美的时序物理映射（与 tr_np 物理时间严格一一对应）
        close_current = close[1:]
        floor_np = np.maximum(close_current * 0.0005, 0.001)

        # 🚨【时间轴动态映射常识锁】
        # 只有当天 TR 超过当天股价的 2 倍，才判定为数据源投毒或乌龙指。完美兼容仙股的历史真实巨震。
        if np.any(tr_np > (close_current * 2.0)):
            return None

        # 🚨【上游相对比例矩阵重整】
        tr_ratio_np = tr_np / close_current

        # 3. 种子期计算与健康度审查
        initial_sma = np.mean(tr_np[:period_days])
        
        # 🚨【无量纲比例流空间种子期拦截】
        if (
            np.isnan(initial_sma) 
            or np.isinf(initial_sma) 
            or np.mean(tr_ratio_np[:period_days]) <= 1e-5
        ):
            return None

        # 分配 ATR 容器
        atr_np = np.zeros(tr_len, dtype=float)
        
        # 🚨【漏洞 1 修正：初始种子双向动态夹逼常识锁】
        # 完美对齐物理时序：close_current[period_days - 1] 严格对应种子期最后一根 K 线的收盘价。
        # 引入 close_current * 2.0 作为硬性天花板，防止历史巨震种子值在后续连续一字跌停时沦为高空挂起的僵尸信号！
        seed_floor = floor_np[period_days - 1]
        seed_ceiling = close_current[period_days - 1] * 2.0
        
        # 使用夹逼法则（Floor <= ATR_Seed <= Ceiling）确立无瑕疵初始波动率
        atr_np[period_days - 1] = max(seed_floor, min(initial_sma, seed_ceiling))

        # 4. 递推期计算（完美的 C-Contiguous 线性自增状态机）
        alpha = 1.0 / period_days
        one_minus_alpha = 1.0 - alpha

        for i in range(period_days, tr_len):
            # 应用相对比例直接拦截一字板假死，消除了循环内的数组寻址与乘法运算
            if tr_ratio_np[i] <= 1e-5:
                # 流动性冻结，状态机挂起，完美强制继承前一日有效的真实波动率
                atr_np[i] = atr_np[i - 1]
            else:
                # 正常交易日，执行 Wilder 平滑迭代并附加硬底线限幅
                raw_atr = alpha * tr_np[i] + one_minus_alpha * atr_np[i - 1]
                atr_np[i] = max(raw_atr, floor_np[i])

        final_atr = atr_np[-1]

        # 5. 终极物理常识防空门
        if final_atr <= 0.0:
            return None

        return float(final_atr)

    except Exception:
        # 实盘高频大并发环境下，使用顶级静默风控熔断，杜绝 I/O 阻塞风险
        return None
